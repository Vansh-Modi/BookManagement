CREATE PROCEDURE [dbo].[AddBooking]
	@UserID INT,
	@CatId INT,
	@MovieId INT,
	@NoOfTickets INT,
	@Amount INT
AS
BEGIN
SET NOCOUNT ON;

	INSERT INTO Tbl_Booking(User_ID, Cat_ID, Movie_ID, no_of_Tickets, amount)
	VALUES(@UserID, @CatId, @MovieId, @NoOfTickets, @Amount);
END



CREATE PROCEDURE [dbo].[AddMovie]
	@Name NVARCHAR(MAX) ,

	@Rate INT

BEGIN

	@booking_ID int

BEGIN

END

CREATE PROCEDURE [dbo].[DisplayBooking]
AS
BEGIN
SET NOCOUNT ON;
	SELECT b.booking_ID AS [booking_ID], 
		   b.User_ID AS [User_ID], 
		   b.Movie_ID AS [Movie_ID], 
		   b.no_of_Tickets AS [no_of_Tickets], 
		   b.Cat_ID AS [Cat_ID], 
		   b.amount AS [amount], 
		   c.Cat_Type AS [Cat_Type], 
		   m.Movie_name AS [movie_name], 
		   m.rate AS [rate], 
		   m.Release_Date AS [date] 
	FROM Tbl_Booking AS b 
	JOIN Tbl_User AS u 
		ON b.User_ID = u.User_ID 
	JOIN Tbl_Movie AS m 
		ON b.Movie_ID = m.Movie_ID 
	JOIN Tbl_Movie_Category AS c 
		ON b.Cat_ID = c.Cat_ID;
END

CREATE PROCEDURE [dbo].[LoginUser]
	@Name int = 0,
	@Pass int
AS
BEGIN 
SET NOCOUNT ON;
	SELECT COUNT(*) AS UserCount FROM Tbl_User WHERE User_Name = @Name AND User_password = @Pass;
END

CREATE PROCEDURE [dbo].[UpdateUser]
	@Name varchar(50), 
	@Email varchar(50),
	@Password varchar(15),
	@City varchar(50),
	@Phone varchar(15),
	@User_ID int
AS
BEGIN
SET NOCOUNT ON;

	UPDATE Tbl_User SET User_name = @Name, Email_ID = @Email, User_password = @Password, City = @City, PhoneNo = @Phone
	WHERE User_ID = @User_ID;
END

CREATE PROCEDURE [dbo].[UpdateMovieCategory]

AS
BEGIN
SET NOCOUNT ON;

END


CREATE PROCEDURE [dbo].[UpdateMovie]
	@CatId int,
	@Name nvarchar(MAX), 
	@Date date,
	@rate int,
	@Movie_ID INT
AS
BEGIN
SET NOCOUNT ON;

	UPDATE Tbl_Movie SET Movie_name = @Name, Release_Date = @Date, rate = @rate, Cat_ID = @CatId
	WHERE Movie_ID = @Movie_ID;
END


CREATE PROCEDURE [dbo].[UpdateBooking]
	@Cat_ID INT,
	@User_ID INT,
	@Movie_ID INT,
	@amount INT,
	@no_of_tickets INT,
	@Booking_ID INT
AS
BEGIN
SET NOCOUNT ON;

	UPDATE Tbl_Booking SET Cat_ID = @Cat_ID, User_ID = @User_ID, Movie_ID = @Movie_ID, amount = @amount, no_of_tickets = @no_of_tickets
	WHERE booking_ID = @Booking_ID;
END



CREATE PROCEDURE [dbo].[DisplayMovies]
@Cat_ID INT = NULL
AS
BEGIN
SET NOCOUNT ON;

	SELECT * FROM Tbl_Movie WHERE (@Cat_ID IS NULL OR Cat_ID = @Cat_ID);
END

CREATE PROCEDURE [dbo].[GetRate]
	@Movie_ID int
AS
SET NOCOUNT ON;
BEGIN
	SELECT rate FROM Tbl_Movie WHERE Movie_ID = @Movie_ID
END
